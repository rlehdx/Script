> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scriva 랜딩 페이지 Hero 섹션을 풀스크린 3D 와이어프레임 배경 + 중앙 글래스모피즘 UI로 업그레이드한다.

`src/components/Scene.tsx`를 아래 내용으로 생성:

```tsx
"use client";

import { useRef, useMemo } from "react";
import { Canvas, useFrame } from "@react-three/fiber";
import { Points, PointMaterial } from "@react-three/drei";
import * as THREE from "three";

// ─── 타입 정의 ────────────────────────────────────────────────────────────────
interface SceneProps {
  mouseX: number;      // -1 ~ 1 정규화된 마우스 X 위치
  mouseY: number;      // -1 ~ 1 정규화된 마우스 Y 위치
  ctaHovered: boolean; // CTA 버튼 hover 상태 (scale 반응용)
}

interface IcosahedronMeshProps {
  mouseX: number;
  mouseY: number;
  ctaHovered: boolean;
}

// ─── 와이어프레임 Icosahedron ──────────────────────────────────────────────────
function IcosahedronMesh({ mouseX, mouseY, ctaHovered }: IcosahedronMeshProps) {
  const meshRef = useRef<THREE.Mesh>(null);

  useFrame((_, delta) => {
    if (!meshRef.current) return;

    // 마우스 위치에 부드럽게 반응 (lerp)
    meshRef.current.rotation.x += (mouseY * 0.5 - meshRef.current.rotation.x) * 0.05;
    meshRef.current.rotation.y += (mouseX * 0.5 - meshRef.current.rotation.y) * 0.05;

    // 기본 자동 회전 (마우스 반응과 합산)
    meshRef.current.rotation.y += delta * 0.15;

    // CTA hover 시 scale 확대 (lerp)
    const targetScale = ctaHovered ? 1.15 : 1.0;
    meshRef.current.scale.setScalar(
      meshRef.current.scale.x + (targetScale - meshRef.current.scale.x) * 0.08
    );
  });

  return (
    <mesh ref={meshRef} position={[2.2, 0, 0]}>
      {/* detail=1 → 적당한 면 수, 성능 최적 */}
      <icosahedronGeometry args={[2.2, 1]} />
      <meshStandardMaterial
        color="#06b6d4"
        wireframe
        emissive="#06b6d4"
        emissiveIntensity={0.15}
        transparent
        opacity={0.7}
      />
    </mesh>
  );
}

// ─── 파티클 시스템 ────────────────────────────────────────────────────────────
function ParticleField() {
  const pointsRef = useRef<THREE.Points>(null);

  // 구 표면에 랜덤 분포된 280개 파티클 좌표 생성
  const positions = useMemo(() => {
    const count = 280;
    const arr = new Float32Array(count * 3);
    for (let i = 0; i < count; i++) {
      // 구 좌표계로 랜덤 분포
      const theta = Math.random() * Math.PI * 2;
      const phi = Math.acos(2 * Math.random() - 1);
      const r = 4 + Math.random() * 3; // 반지름 4~7
      arr[i * 3] = r * Math.sin(phi) * Math.cos(theta);
      arr[i * 3 + 1] = r * Math.sin(phi) * Math.sin(theta);
      arr[i * 3 + 2] = r * Math.cos(phi);
    }
    return arr;
  }, []);

  // 파티클 느린 자체 회전
  useFrame((_, delta) => {
    if (!pointsRef.current) return;
    pointsRef.current.rotation.y += delta * 0.03;
    pointsRef.current.rotation.x += delta * 0.01;
  });

  return (
    <Points ref={pointsRef} positions={positions} stride={3}>
      <PointMaterial
        color="#a78bfa"
        size={0.04}
        sizeAttenuation
        transparent
        opacity={0.6}
        depthWrite={false}
      />
    </Points>
  );
}

// ─── 씬 내부 조명 + 오브젝트 ──────────────────────────────────────────────────
function SceneContent({ mouseX, mouseY, ctaHovered }: SceneProps) {
  return (
    <>
      {/* Ambient — 전체 밝기 기반 */}
      <ambientLight intensity={0.3} />
      {/* Cyan 포인트 라이트 — 우측 상단 */}
      <pointLight position={[10, 10, 10]} color="#06b6d4" intensity={1.5} />
      {/* Purple 포인트 라이트 — 좌측 하단 */}
      <pointLight position={[-10, -5, -5]} color="#7c3aed" intensity={1.0} />
      {/* Directional — 전면 보조 */}
      <directionalLight position={[0, 5, 5]} intensity={0.5} />

      <IcosahedronMesh mouseX={mouseX} mouseY={mouseY} ctaHovered={ctaHovered} />
      <ParticleField />
    </>
  );
}

// ─── 메인 내보내기: Canvas 래퍼 ───────────────────────────────────────────────
export default function Scene({ mouseX, mouseY, ctaHovered }: SceneProps) {
  return (
    <Canvas
      // 투명 배경으로 CSS 배경과 합성
      gl={{ antialias: true, alpha: true }}
      // 마우스/스크롤 이벤트 발생 시에만 재렌더 → 성능 절약
      frameloop="always"
      // 디바이스 픽셀비 최대 1.5로 캡 (고해상도 디바이스 성능 보호)
      dpr={[1, 1.5]}
      camera={{ position: [0, 0, 8], fov: 50 }}
      style={{ position: "absolute", inset: 0 }}
    >
      <SceneContent mouseX={mouseX} mouseY={mouseY} ctaHovered={ctaHovered} />
    </Canvas>
  );
}
```
